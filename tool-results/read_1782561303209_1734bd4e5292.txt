     8→     8→import {

    18→    18→import { Filter } from "lucide-react";
    19→    19→import type { PageId, NavContext } from "../kj/header";

    22→    22→
    23→    23→interface ShopPageProps {

    29→    29→const PER_PAGE = 24;
    30→    30→
    31→    31→export function ShopPage({ initialFilters, onNavigate }: ShopPageProps) {
    32→    32→  const { t, lang } = useLang();
    33→    33→  // Filter state
    34→    34→  const [activeCategory, setActiveCategory] = useState<string>(initialFilters?.category ?? "alle");
    35→    35→  const [activeSubcategory, setActiveSubcategory] = useState<string>(initialFilters?.subcategory ?? "alle");
    36→    36→  const [activeBrand, setActiveBrand] = useState<string>(initialFilters?.brand ?? "alle");
    37→    37→  const [searchQ, setSearchQ] = useState<string>(initialFilters?.q ?? "");
    38→    38→  const [priceRange, setPriceRange] = useState<[number, number]>([0, 25000]);
    39→    39→  const [inStockOnly, setInStockOnly] = useState(false);
    40→    40→
    41→    41→  // Sort + pagination
    42→    42→  const [sort, setSort] = useState<SortKey>("recommended");
    43→    43→  const [page, setPage] = useState(1);
    44→    44→
    45→    45→  // Brand filter — show top 20 by default with a "show more" toggle
    46→    46→  const [showAllBrands, setShowAllBrands] = useState(false);
    47→    47→
    48→    48→  // Data
    49→    49→  const [products, setProducts] = useState<Product[] | null>(null);
    50→    50→  const [totalPages, setTotalPages] = useState(1);
    51→    51→  const [totalCount, setTotalCount] = useState(0);
    52→    52→  const [categories, setCategories] = useState<CategoryNode[]>([]);
    53→    53→  const [brands, setBrands] = useState<BrandNode[]>([]);
    54→    54→
    55→    55→  // Sync URL-driven initial filters into state when they change
    56→    56→  useEffect(() => {
    57→    57→    if (initialFilters) {
    58→    58→      if (initialFilters.category !== undefined) setActiveCategory(initialFilters.category);
    59→    59→      if (initialFilters.subcategory !== undefined) setActiveSubcategory(initialFilters.subcategory);
    60→    60→      if (initialFilters.brand !== undefined) setActiveBrand(initialFilters.brand);
    61→    61→      if (initialFilters.q !== undefined) setSearchQ(initialFilters.q);
    62→    62→      setPage(1);
    63→    63→    }
    64→    64→     
    65→    65→  }, [JSON.stringify(initialFilters)]);
    66→    66→
    67→    67→  // Fetch categories + brands once

    73→    73→          fetch("/api/categories", { cache: "no-store" }),
    74→    74→          fetch("/api/brands", { cache: "no-store" }),

    78→    78→          const data = await catRes.json();
    79→    79→          setCategories(data.categories ?? []);
    80→    80→        }
    81→    81→        if (brandRes.ok) {
    82→    82→          const data = await brandRes.json();
    83→    83→          setBrands((data.brands ?? []).filter((b: BrandNode) => b.count > 0));
    84→    84→        }
    85→    85→      } catch {
    86→    86→        /* ignore */
    87→    87→      }
    88→    88→    })();
    89→    89→    return () => {
    90→    90→      cancelled = true;
    91→    91→    };
    92→    92→  }, []);
    93→    93→
    94→    94→  // Fetch products whenever filters/sort/page change (debounced via useEffect)

    96→    96→    setProducts(null); // show skeletons during fetch
    97→    97→    try {
    98→    98→      const params = new URLSearchParams();
    99→    99→      if (activeCategory !== "alle") params.set("category", activeCategory);
   100→   100→      if (activeSubcategory !== "alle") params.set("subcategory", activeSubcategory);
   101→   101→      if (activeBrand !== "alle") params.set("brand", activeBrand);
   102→   102→      if (searchQ.trim()) params.set("q", searchQ.trim());
   103→   103→      params.set("minPrice", String(priceRange[0]));
   104→   104→      params.set("maxPrice", String(priceRange[1]));

   106→   106→      params.set("sort", sort);
   107→   107→      params.set("page", String(page));
   108→   108→      params.set("perPage", String(PER_PAGE));
   109→   109→      params.set("includeCount", "1");

   111→   111→      const res = await fetch(`/api/products?${params.toString()}`, { cache: "no-store" });
   112→   112→      if (!res.ok) throw new Error("Kunne ikke hente produkter");

   115→   115→      setTotalPages(data.totalPages);
   116→   116→      setTotalCount(data.totalCount ?? data.products.length);

   118→   118→      setProducts([]);
   119→   119→    }
   120→   120→  }, [activeCategory, activeSubcategory, activeBrand, searchQ, priceRange, inStockOnly, sort, page]);
   121→   121→
   122→   122→  useEffect(() => {
   123→   123→    void fetchProducts();
   124→   124→  }, [fetchProducts]);
   125→   125→
   126→   126→  // Reset to page 1 when filters change
   127→   127→  useEffect(() => {
   128→   128→    setPage(1);
   129→   129→  }, [activeCategory, activeSubcategory, activeBrand, searchQ, priceRange, inStockOnly, sort]);
   130→   130→
   131→   131→  const activeCategoryObj = categories.find((c) => c.slug === activeCategory);
   132→   132→  // When only a subcategory is selected (e.g. from the mega menu), find its
   133→   133→  // parent category so we can show the right heading + subcategory breadcrumb.
   134→   134→  const activeSubcategoryObj = !activeCategoryObj && activeSubcategory !== "alle"
   135→   135→    ? categories
   136→   136→        .flatMap((c) => c.subcategories.map((s) => ({ ...s, parent: c })))
   137→   137→        .find((s) => s.slug === activeSubcategory)
   138→   138→    : activeCategoryObj?.subcategories.find((s) => s.slug === activeSubcategory);
   139→   139→  const parentForHeading = activeCategoryObj ?? (activeSubcategoryObj as { parent?: CategoryNode } | undefined)?.parent;
   140→   140→  const isOutlet = activeCategory === "outlet";
   141→   141→
   142→   142→  // Auto-switch sort to "discount" when entering Outlet (per merchandising strategy)
   143→   143→  const effectiveSort: SortKey = isOutlet && sort === "recommended" ? "discount" : sort;
   144→   144→
   145→   145→  // Available sort options: hide "discount" unless in Outlet; show all others always
   146→   146→  const visibleSortOptions = SORT_OPTIONS.filter((o) => {
   147→   147→    // Hide "discount" unless we're in Outlet — no products have
   148→   148→    // originalPrice set yet (catalog has no prices).
   149→   149→    if (o.value === "discount") return isOutlet;
   150→   150→    // Hide price-based sorts while the catalog has no prices imported.
   151→   151→    // (All products have price=0 so these would be no-ops.)
   152→   152→    if (o.value === "price_asc" || o.value === "price_desc") return false;
   153→   153→    return true;
   154→   154→  });
   155→   155→
   156→   156→  // Reset subcategory when changing main category
   157→   157→  const handleCategoryClick = (slug: string) => {

   159→   159→    setActiveSubcategory("alle");
   160→   160→  };
   161→   161→
   162→   162→  // Active filter count for the mobile filter sheet badge
   163→   163→  const activeFilterCount =
   164→   164→    (activeCategory !== "alle" ? 1 : 0) +
   165→   165→    (activeSubcategory !== "alle" ? 1 : 0) +
   166→   166→    (activeBrand !== "alle" ? 1 : 0) +
   167→   167→    (inStockOnly ? 1 : 0) +
   168→   168→    (priceRange[0] > 0 || priceRange[1] < 25000 ? 1 : 0);
   169→   169→
   170→   170→  const clearFilters = () => {
   171→   171→    setActiveCategory("alle");
   172→   172→    setActiveSubcategory("alle");
   173→   173→    setActiveBrand("alle");
   174→   174→    setSearchQ("");
   175→   175→    setPriceRange([0, 25000]);
   176→   176→    setInStockOnly(false);
   177→   177→  };
   178→   178→
   179→   179→  // ===== Filter UI =====
   180→   180→  const FiltersPanel = (

   182→   182→      {/* Category */}
   183→   183→      <div>
   184→   184→        <h4 className="mb-3 text-[11px] font-semibold uppercase tracking-[0.15em] text-[#8a96a1]">
   185→   185→          Kategori
   186→   186→        </h4>
   187→   187→        <div className="flex flex-col gap-1.5">
   188→   188→          <FilterRadio
   189→   189→            label="Alle kategorier"
   190→   190→            checked={activeCategory === "alle"}
   191→   191→            onChange={() => handleCategoryClick("alle")}
   192→   192→          />
   193→   193→          {categories.map((c) => (
   194→   194→            <FilterRadio
   195→   195→              key={c.id}
   196→   196→              label={`${c.name} (${c.count})`}
   197→   197→              checked={activeCategory === c.slug}
   198→   198→              onChange={() => handleCategoryClick(c.slug)}
   199→   199→            />
   200→   200→          ))}
   201→   201→        </div>
   202→   202→      </div>
   203→   203→
   204→   204→      {/* Subcategory filter — show when a parent category with subs is resolved */}
   205→   205→      {parentForHeading && parentForHeading.subcategories.filter((s) => s.count > 0).length > 0 && (
   206→   206→        <div>
   207→   207→          <h4 className="mb-3 text-[11px] font-semibold uppercase tracking-[0.15em] text-[#8a96a1]">
   208→   208→            {t("shop.subcategory")}
   209→   209→          </h4>
   210→   210→          <div className="flex flex-col gap-1.5 max-h-60 overflow-y-auto kj-scroll pr-2">

   213→   213→              checked={activeSubcategory === "alle"}
   214→   214→              onChange={() => setActiveSubcategory("alle")}
   215→   215→            />
   216→   216→            {parentForHeading.subcategories
   217→   217→              .filter((s) => s.count > 0)
   218→   218→              .map((sub) => (
   219→   219→                <FilterRadio
   220→   220→                  key={sub.id}
   221→   221→                  label={`${sub.name} (${sub.count})`}
   222→   222→                  checked={activeSubcategory === sub.slug}
   223→   223→                  onChange={() => setActiveSubcategory(sub.slug)}
   224→   224→                />
   225→   225→              ))}
   226→   226→          </div>
   227→   227→        </div>
   228→   228→      )}
   229→   229→
   230→   230→      {/* Brand — top 20 with "show more" for the long tail */}
   231→   231→      <div>
   232→   232→        <h4 className="mb-3 text-[11px] font-semibold uppercase tracking-[0.15em] text-[#8a96a1]">
   233→   233→          {t("shop.brand")}
   234→   234→        </h4>
   235→   235→        <div className="flex flex-col gap-1.5 max-h-60 overflow-y-auto kj-scroll pr-2">
   236→   236→          <FilterRadio
   237→   237→            label={t("shop.allBrands")}
   238→   238→            checked={activeBrand === "alle"}
   239→   239→            onChange={() => setActiveBrand("alle")}
   240→   240→          />
   241→   241→          {(showAllBrands ? brands : brands.slice(0, 20)).map((b) => (
   242→   242→            <FilterRadio
   243→   243→              key={b.id}
   244→   244→              label={`${b.name} (${b.count})`}
   245→   245→              checked={activeBrand === b.slug}
   246→   246→              onChange={() => setActiveBrand(b.slug)}
   247→   247→            />
   248→   248→          ))}
   249→   249→          {brands.length > 20 && (
   250→   250→            <button
   251→   251→              onClick={() => setShowAllBrands((v) => !v)}
   252→   252→              className="mt-1 text-[11px] font-semibold uppercase tracking-[0.1em] text-[#c75d2c] hover:underline"
   253→   253→            >
   254→   254→              {showAllBrands ? (lang === "no" ? "Vis mindre" : "Show less") : `${lang === "no" ? "Vis alle" : "Show all"} (${brands.length})`}
   255→   255→            </button>
   256→   256→          )}
   257→   257→        </div>
   258→   258→      </div>
   259→   259→
   260→   260→      {/* Price range — hidden for now because the real Kleven catalog
   261→   261→          doesn't ship prices (priceNok=0 for all products). The slider UI
   262→   262→          stays in the source so it can be re-enabled once prices are
   263→   263→          imported from the upstream product pages. */}
   264→   264→      {false && (
   265→   265→      <div>
   266→   266→        <h4 className="mb-3 text-[11px] font-semibold uppercase tracking-[0.15em] text-[#8a96a1]">
   267→   267→          Pris (kr)
   268→   268→        </h4>
   269→   269→        <div className="px-1">
   270→   270→          <Slider
   271→   271→            value={priceRange}
   272→   272→            min={0}
   273→   273→            max={25000}
   274→   274→            step={100}
   275→   275→            onValueChange={(v) => setPriceRange([v[0], v[1]] as [number, number])}
   276→   276→            className="my-2"
   277→   277→          />
   278→   278→          <div className="flex items-center justify-between text-[12px] text-[#3a4856]">
   279→   279→            <span>Kr {priceRange[0].toLocaleString("no-NO")}</span>
   280→   280→            <span>Kr {priceRange[1].toLocaleString("no-NO")}+</span>
   281→   281→          </div>
   282→   282→        </div>
   283→   283→      </div>
   284→   284→      )}
   285→   285→
   286→   286→      {/* Price notice */}
   287→   287→      <div className="rounded-md border border-[#f0c548]/40 bg-[#f0c548]/10 px-3 py-2 text-[11px] leading-relaxed text-[#3a4856]">
   288→   288→        {lang === "no"
   289→   289→          ? "Priser kommer — Kleven-katalogen oppdateres for øyeblikket. Bruk telefonnummeret på produktsiden for direkte pris forespørsel."
   290→   290→          : "Prices coming soon — the Kleven catalog is currently being updated. Use the phone number on the product page for direct price inquiries."}
   291→   291→      </div>
   292→   292→
   293→   293→      {/* In stock */}
   294→   294→      <label className="flex cursor-pointer items-center gap-2 text-[13px] text-[#1f2d3a]">
   295→   295→        <input
   296→   296→          type="checkbox"
   297→   297→          checked={inStockOnly}
   298→   298→          onChange={(e) => setInStockOnly(e.target.checked)}
   299→   299→          className="h-4 w-4 accent-[#1f2d3a]"
   300→   300→        />
   301→   301→        {t("shop.inStockOnly")}
   302→   302→      </label>
   303→   303→
   304→   304→      {activeFilterCount > 0 && (
   305→   305→        <button
   306→   306→          onClick={clearFilters}
   307→   307→          className="flex items-center gap-1 text-[12px] font-semibold text-[#c75d2c] hover:underline"
   308→   308→        >
   309→   309→          <X size={12} /> Tøm filtre ({activeFilterCount})
   310→   310→        </button>
   311→   311→      )}
   312→   312→    </div>
   313→   313→  );
   314→   314→
   315→   315→  return (
   316→   316→    <div className="kj-page-enter">
   317→   317→      <ShippingBanner size="md" />
   318→   318→
   319→   319→      {/* Shop header */}
   320→   320→      <section className="w-full" style={{ backgroundColor: "#e9e5db" }}>
   321→   321→        <div className="mx-auto max-w-[1280px] px-6 pt-12 pb-6 lg:px-10">
   322→   322→          {searchQ && (
   323→   323→            <div className="mb-2 text-[11px] font-semibold uppercase tracking-[0.2em] text-[#8a96a1]">
   324→   324→              <span className="normal-case text-[#1f2d3a]">
   325→   325→                {t("nav.search")}: &laquo;{searchQ}&raquo;
   326→   326→              </span>
   327→   327→            </div>
   328→   328→          )}
   329→   329→          <h1
   330→   330→            className="text-[clamp(2.25rem,4.5vw,3.25rem)] font-bold tracking-[-0.02em] text-[#1f2d3a]"
   331→   331→            style={{ fontFamily: "var(--font-montserrat), sans-serif" }}
   332→   332→          >
   333→   333→            {parentForHeading
   334→   334→              ? parentForHeading.name
   335→   335→              : searchQ
   336→   336→                ? `${t("nav.search")}: "${searchQ}"`
   337→   337→                : t("shop.title")}
   338→   338→          </h1>
   339→   339→          {activeSubcategoryObj && (
   340→   340→            <p className="mt-2 text-[16px] font-medium uppercase tracking-[0.1em] text-[#2d4a3e]">
   341→   341→              {activeSubcategoryObj.name}
   342→   342→            </p>
   343→   343→          )}
   344→   344→          <p className="mt-3 max-w-xl text-[14px] font-light leading-relaxed text-[#6b7884]">
   345→   345→            {t("shop.desc")}
   346→   346→          </p>
   347→   347→

   350→   350→            <CategoryPill
   351→   351→              label={t("shop.all")}
   352→   352→              active={activeCategory === "alle"}
   353→   353→              onClick={() => handleCategoryClick("alle")}
   354→   354→            />
   355→   355→            {categories.map((c) => (
   356→   356→              <CategoryPill
   357→   357→                key={c.id}
   358→   358→                label={c.name}
   359→   359→                active={activeCategory === c.slug}
   360→   360→                onClick={() => handleCategoryClick(c.slug)}
   361→   361→              />
   362→   362→            ))}
   363→   363→          </div>
   364→   364→
   365→   365→          {/* Subcategory pills — show when a category with subs is selected */}
   366→   366→          {parentForHeading && parentForHeading.subcategories.filter((s) => s.count > 0).length > 0 && (
   367→   367→            <div className="mt-4 flex flex-wrap items-center gap-1.5 border-b border-[#d4cfc1] pb-5">

   370→   370→              </span>

   376→   376→              {parentForHeading.subcategories
   377→   377→                .filter((s) => s.count > 0)
   378→   378→                .map((sub) => (
   379→   379→                  <SubcategoryPill
   380→   380→                    key={sub.id}
   381→   381→                    label={sub.name}

   393→   393→              {totalCount !== 1 ? t("shop.articles") : t("shop.article")}
   394→   394→              {parentForHeading && (
   395→   395→                <>
   396→   396→                  {" "}{t("shop.in")} <span className="font-semibold text-[#1f2d3a]">{parentForHeading.name}</span>
   397→   397→                </>
   398→   398→              )}
   399→   399→              {activeSubcategoryObj && (

   401→   401→                  {" "}→ <span className="font-semibold text-[#2d4a3e]">{activeSubcategoryObj.name}</span>

   404→   404→            </div>
   405→   405→
   406→   406→            <div className="flex items-center gap-3">
   407→   407→              {/* Mobile filter button */}
   408→   408→              <Sheet>

   541→   541→                              setPage(n);
   542→   542→                              window.scrollTo({ top: 0, behavior: "smooth" });
   543→   543→                            }}
   544→   544→                            className={`rounded-full px-3 py-1.5 text-[12px] font-semibold transition-colors ${
   545→   545→                              n === page
   546→   546→                                ? "bg-[#1f2d3a] text-white"
   547→   547→                                : "text-[#6b7884] hover:bg-white hover:text-[#1f2d3a]"
   548→   548→                            }`}
   549→   549→                          >
   550→   550→                            {n}
   551→   551→                          </button>
   552→   552→                        </span>
   553→   553→                      );
   554→   554→                    })}
   555→   555→
   556→   556→                  <button
   557→   557→                    aria-label="Neste side"
   558→   558→                    disabled={page === totalPages}
   559→   559→                    onClick={() => {
   560→   560→                      setPage((p) => Math.min(totalPages, p + 1));
   561→   561→                      window.scrollTo({ top: 0, behavior: "smooth" });
   562→   562→                    }}
   563→   563→                    className="flex h-9 w-9 items-center justify-center rounded-full border border-[#d4cfc1] bg-white text-[#1f2d3a] transition-colors hover:bg-[#1f2d3a] hover:text-white disabled:opacity-40 disabled:hover:bg-white disabled:hover:text-[#1f2d3a]"
   564→   564→                  >
   565→   565→                    <ChevronRight size={16} />
   566→   566→                  </button>
   567→   567→                </div>
   568→   568→              )}
   569→   569→            </div>
   570→   570→          </div>
   571→   571→        </div>
   572→   572→      </section>
   573→   573→    </div>
   574→   574→  );
   575→   575→}
