  1359→/**
  1360→ * Compute the "Recommended" merchandising score for a product.
  1361→ * Weighted blend of revenue proxy, conversion rate, inventory health,
  1362→ * popularity, seasonality, and margin. Returns 0-100.
  1363→ *
  1364→ * Score =
  1365→ *   (Revenue Weight × 30%)    — sales90 × priceNok (normalized)
  1366→ * + (Conversion Rate × 25%)
  1367→ * + (Inventory Health × 15%)
  1368→ * + (Popularity × 15%)
  1369→ * + (Seasonality × 10%)
  1370→ * + (Margin × 5%)
  1371→ */
  1372→function recommendedScore(p: Product): number {
  1373→  // Revenue proxy: normalize sales90 × priceNok to 0-1
  1374→  const revenue = p.sales90 * p.priceNok;
  1375→  const maxRevenue = 4_000_000; // approx max in catalog
  1376→  const revenueNorm = Math.min(revenue / maxRevenue, 1);
  1377→
  1378→  const score =
  1379→    revenueNorm * 30 +
  1380→    p.conversionRate * 10 * 25 + // scale 0-0.2 → 0-50, then ×25%
  1381→    p.inventoryHealth * 15 +
  1382→    (p.popularity / 100) * 15 +
  1383→    p.seasonBoost * 10 +
  1384→    p.margin * 5;
  1385→
  1386→  // Inventory penalty: stock = 0 → -20, stock = 1 → -10
  1387→  let penalty = 0;
  1388→  if (p.stockCount === 0) penalty = -20;
  1389→  else if (p.stockCount === 1) penalty = -10;
  1390→  else if (p.stockCount > 20) penalty = 5; // boost well-stocked items
  1391→
  1392→  return Math.max(0, Math.min(100, score + penalty));
  1393→}
  1394→
  1395→/**
  1396→ * Discount percentage for a product (0-1). 0 if no original price.
  1397→ */
  1398→function discountPct(p: Product): number {
  1399→  if (!p.originalNok || p.originalNok <= p.priceNok) return 0;
  1400→  return (p.originalNok - p.priceNok) / p.originalNok;
  1401→}
